// Global state management
class JobScheduler {
    constructor() {
        this.jobs = new Map();
        this.timers = new Map();
        this.currentScript = null;
        this.init();
    }

    init() {
        this.loadJobs();
        this.bindEvents();
        this.setDefaultDateTime();
        this.updateJobsCount();
        this.scheduleExistingJobs();
    }

    // In-memory storage methods (simulating localStorage)
    loadJobs() {
        // In a real app, this would be: JSON.parse(localStorage.getItem('scheduledJobs') || '[]')
        // For demo purposes, we'll start with empty jobs
        this.jobs = new Map();
    }

    saveJobs() {
        // In a real app, this would be: localStorage.setItem('scheduledJobs', JSON.stringify(Array.from(this.jobs.values())))
        // For demo, we just keep jobs in memory
        console.log('Jobs saved to memory:', Array.from(this.jobs.values()));
    }

    // Event binding
    bindEvents() {
        // Tab navigation
        document.querySelectorAll('.nav-tab').forEach(tab => {
            tab.addEventListener('click', (e) => this.switchTab(e.target.dataset.tab));
        });

        // File upload
        const fileInput = document.getElementById('script-file');
        const uploadForm = document.getElementById('upload-form');
        const schedulingForm = document.getElementById('scheduling-form');

        fileInput.addEventListener('change', (e) => this.handleFileSelect(e));
        uploadForm.addEventListener('submit', (e) => this.handleUpload(e));
        schedulingForm.addEventListener('submit', (e) => this.handleScheduling(e));

        // Modal close
        document.querySelector('.modal-close').addEventListener('click', () => this.closeModal());
        document.getElementById('log-modal').addEventListener('click', (e) => {
            if (e.target.id === 'log-modal') this.closeModal();
        });

        // Search and filter
        document.getElementById('search-jobs').addEventListener('input', (e) => this.filterJobs());
        document.getElementById('filter-status').addEventListener('change', (e) => this.filterJobs());

        // Run immediately checkbox
        document.getElementById('run-immediately').addEventListener('change', (e) => {
            const startTime = document.getElementById('start-time');
            if (e.target.checked) {
                startTime.value = new Date().toISOString().slice(0, 16);
            }
        });
    }

    // Tab switching
    switchTab(tabName) {
        document.querySelectorAll('.nav-tab').forEach(tab => tab.classList.remove('active'));
        document.querySelectorAll('.tab-content').forEach(content => content.classList.remove('active'));
        
        document.querySelector(`[data-tab="${tabName}"]`).classList.add('active');
        document.getElementById(`${tabName}-tab`).classList.add('active');

        if (tabName === 'jobs') {
            this.renderJobs();
        }
    }

    // File handling
    handleFileSelect(event) {
        const file = event.target.files[0];
        const fileError = document.getElementById('file-error');
        const uploadBtn = document.getElementById('upload-btn');
        const fileDisplay = document.querySelector('.file-input-display');
        const scriptName = document.getElementById('script-name');
        const platform = document.getElementById('platform');

        fileError.classList.add('hidden');

        if (!file) {
            uploadBtn.disabled = true;
            fileDisplay.classList.remove('has-file');
            return;
        }

        // Validate file
        const validExtensions = ['.sh', '.bat'];
        const fileName = file.name.toLowerCase();
        const isValidExtension = validExtensions.some(ext => fileName.endsWith(ext));
        const maxSize = 1024 * 1024; // 1MB

        if (!isValidExtension) {
            this.showError(fileError, 'Please select a .sh or .bat file');
            uploadBtn.disabled = true;
            return;
        }

        if (file.size > maxSize) {
            this.showError(fileError, 'File size must be less than 1MB');
            uploadBtn.disabled = true;
            return;
        }

        // Auto-fill script name and platform
        const baseName = file.name.replace(/\.[^/.]+$/, '');
        scriptName.value = baseName;
        platform.value = fileName.endsWith('.bat') ? 'windows' : 'linux';

        // Update UI
        fileDisplay.classList.add('has-file');
        fileDisplay.querySelector('.file-text').textContent = file.name;
        uploadBtn.disabled = false;
    }

    // Upload handling
    async handleUpload(event) {
        event.preventDefault();
        
        const fileInput = document.getElementById('script-file');
        const scriptName = document.getElementById('script-name').value.trim();
        const platform = document.getElementById('platform').value;

        if (!fileInput.files[0] || !scriptName) {
            this.showToast('Please fill in all required fields', 'error');
            return;
        }

        this.showLoading(true);

        try {
            const fileContent = await this.readFile(fileInput.files[0]);
            
            this.currentScript = {
                name: scriptName,
                platform: platform,
                content: fileContent,
                filename: fileInput.files[0].name
            };

            // Show scheduling section
            document.getElementById('scheduling-section').classList.remove('hidden');
            this.showToast('Script uploaded successfully!', 'success');
            
        } catch (error) {
            this.showToast('Error reading file: ' + error.message, 'error');
        }

        this.showLoading(false);
    }

    // File reading utility
    readFile(file) {
        return new Promise((resolve, reject) => {
            const reader = new FileReader();
            reader.onload = e => resolve(e.target.result);
            reader.onerror = e => reject(new Error('Failed to read file'));
            reader.readAsText(file);
        });
    }

    // Scheduling handling
    handleScheduling(event) {
        event.preventDefault();

        if (!this.currentScript) {
            this.showToast('No script uploaded', 'error');
            return;
        }

        const frequencyValue = parseInt(document.getElementById('frequency-value').value);
        const frequencyUnit = document.getElementById('frequency-unit').value;
        const startTime = document.getElementById('start-time').value;
        const endTime = document.getElementById('end-time').value;
        const runImmediately = document.getElementById('run-immediately').checked;

        if (!startTime) {
            this.showToast('Please set a start time', 'error');
            return;
        }

        const startDate = new Date(startTime);
        const endDate = endTime ? new Date(endTime) : null;

        if (startDate <= new Date() && !runImmediately) {
            this.showToast('Start time must be in the future', 'error');
            return;
        }

        if (endDate && endDate <= startDate) {
            this.showToast('End time must be after start time', 'error');
            return;
        }

        // Create job
        const job = {
            id: this.generateId(),
            name: this.currentScript.name,
            platform: this.currentScript.platform,
            content: this.currentScript.content,
            filename: this.currentScript.filename,
            frequency: {
                value: frequencyValue,
                unit: frequencyUnit
            },
            startTime: startDate.toISOString(),
            endTime: endDate ? endDate.toISOString() : null,
            status: 'pending',
            lastRun: null,
            nextRun: runImmediately ? new Date().toISOString() : startDate.toISOString(),
            logs: [],
            createdAt: new Date().toISOString()
        };

        this.jobs.set(job.id, job);
        this.saveJobs();
        this.scheduleJob(job);
        this.updateJobsCount();

        this.showToast(`Job "${job.name}" scheduled successfully!`, 'success');
        this.resetForms();
    }

    // Job scheduling engine
    scheduleJob(job) {
        if (this.timers.has(job.id)) {
            clearTimeout(this.timers.get(job.id));
        }

        const nextRun = new Date(job.nextRun);
        const now = new Date();
        const delay = Math.max(0, nextRun.getTime() - now.getTime());

        const timer = setTimeout(() => {
            this.executeJob(job.id);
        }, delay);

        this.timers.set(job.id, timer);
    }

    // Job execution simulation
    async executeJob(jobId) {
        const job = this.jobs.get(jobId);
        if (!job) return;

        // Check if job should still run
        if (job.endTime && new Date() > new Date(job.endTime)) {
            job.status = 'completed';
            this.jobs.set(jobId, job);
            this.saveJobs();
            return;
        }

        job.status = 'running';
        job.lastRun = new Date().toISOString();
        this.jobs.set(jobId, job);
        this.saveJobs();

        // Simulate script execution
        const executionLog = await this.simulateScriptExecution(job);
        
        // Update job with results
        job.status = executionLog.success ? 'success' : 'error';
        job.logs.push(executionLog);
        
        // Calculate next run
        const nextRun = this.calculateNextRun(job);
        if (nextRun && (!job.endTime || nextRun < new Date(job.endTime))) {
            job.nextRun = nextRun.toISOString();
            job.status = 'pending';
            this.scheduleJob(job);
        } else {
            job.nextRun = null;
            job.status = 'completed';
        }

        this.jobs.set(jobId, job);
        this.saveJobs();

        // Show notification
        const message = executionLog.success ? 
            `Job "${job.name}" executed successfully` : 
            `Job "${job.name}" failed with errors`;
        this.showToast(message, executionLog.success ? 'success' : 'error');
    }

    // Script execution simulation
    async simulateScriptExecution(job) {
        const timestamp = new Date().toISOString();
        const lines = job.content.split('\n').filter(line => line.trim());
        let output = `[${timestamp}] Starting execution of ${job.filename}\n`;
        let success = true;

        // Simulate execution delay
        await new Promise(resolve => setTimeout(resolve, 1000 + Math.random() * 2000));

        for (const line of lines) {
            if (line.trim().startsWith('#') || line.trim().startsWith('REM')) {
                // Skip comments
                continue;
            }
            
            // Simulate command execution
            output += `> ${line}\n`;
            
            // Randomly simulate some failures for demo
            if (Math.random() < 0.1 && line.trim() !== '') {
                output += `Error: Command failed with exit code 1\n`;
                success = false;
                break;
            } else {
                // Simulate successful output
                output += `Command executed successfully\n`;
            }
        }

        output += `[${new Date().toISOString()}] Execution ${success ? 'completed' : 'failed'}\n`;

        return {
            timestamp,
            output,
            success,
            duration: Math.floor(1000 + Math.random() * 2000)
        };
    }

    // Calculate next run time
    calculateNextRun(job) {
        const lastRun = new Date(job.lastRun);
        const { value, unit } = job.frequency;
        
        let nextRun = new Date(lastRun);
        
        switch (unit) {
            case 'minutes':
                nextRun.setMinutes(nextRun.getMinutes() + value);
                break;
            case 'hours':
                nextRun.setHours(nextRun.getHours() + value);
                break;
            case 'days':
                nextRun.setDate(nextRun.getDate() + value);
                break;
        }

        return nextRun;
    }

    // Schedule existing jobs on page load
    scheduleExistingJobs() {
        this.jobs.forEach(job => {
            if (job.status === 'pending' && job.nextRun) {
                const nextRun = new Date(job.nextRun);
                if (nextRun > new Date()) {
                    this.scheduleJob(job);
                } else {
                    // Missed execution, schedule immediately
                    job.nextRun = new Date().toISOString();
                    this.scheduleJob(job);
                }
            }
        });
    }

    // Job management
    cancelJob(jobId) {
        const job = this.jobs.get(jobId);
        if (!job) return;

        if (this.timers.has(jobId)) {
            clearTimeout(this.timers.get(jobId));
            this.timers.delete(jobId);
        }

        job.status = 'cancelled';
        job.endTime = new Date().toISOString();
        job.nextRun = null;
        
        this.jobs.set(jobId, job);
        this.saveJobs();
        this.renderJobs();
        this.showToast(`Job "${job.name}" cancelled`, 'success');
    }

    deleteJob(jobId) {
        const job = this.jobs.get(jobId);
        if (!job) return;

        if (this.timers.has(jobId)) {
            clearTimeout(this.timers.get(jobId));
            this.timers.delete(jobId);
        }

        this.jobs.delete(jobId);
        this.saveJobs();
        this.updateJobsCount();
        this.renderJobs();
        this.showToast(`Job "${job.name}" deleted`, 'success');
    }

    viewLogs(jobId) {
        const job = this.jobs.get(jobId);
        if (!job) return;

        const modal = document.getElementById('log-modal');
        const title = document.getElementById('modal-title');
        const content = document.getElementById('log-content');

        title.textContent = `Logs for ${job.name}`;
        
        if (job.logs.length === 0) {
            content.textContent = 'No execution logs available yet.';
        } else {
            content.textContent = job.logs.map(log => log.output).join('\n---\n\n');
        }

        modal.classList.remove('hidden');
    }

    closeModal() {
        document.getElementById('log-modal').classList.add('hidden');
    }

    // Job rendering
    renderJobs() {
        const jobsList = document.getElementById('jobs-list');
        const jobs = Array.from(this.jobs.values());

        if (jobs.length === 0) {
            jobsList.innerHTML = `
                <div class="no-jobs">
                    <span class="no-jobs-icon">📝</span>
                    <p>No jobs scheduled yet</p>
                    <p class="no-jobs-subtitle">Upload and schedule your first script to get started</p>
                </div>
            `;
            return;
        }

        const filteredJobs = this.getFilteredJobs(jobs);

        if (filteredJobs.length === 0) {
            jobsList.innerHTML = `
                <div class="no-jobs">
                    <span class="no-jobs-icon">🔍</span>
                    <p>No jobs match your filters</p>
                </div>
            `;
            return;
        }

        jobsList.innerHTML = filteredJobs.map(job => this.renderJobCard(job)).join('');
    }

    renderJobCard(job) {
        const platformIcon = job.platform === 'windows' ? '🪟' : '🐧';
        const statusClass = `status-${job.status}`;
        const nextRunText = job.nextRun ? 
            this.formatDateTime(job.nextRun) : 
            'Not scheduled';
        const lastRunText = job.lastRun ? 
            this.formatDateTime(job.lastRun) : 
            'Never';

        return `
            <div class="job-card">
                <div class="job-header">
                    <div class="job-title">
                        <span class="platform-icon">${platformIcon}</span>
                        <h3>${job.name}</h3>
                    </div>
                    <div class="job-status">
                        <span class="status-badge ${statusClass}">${job.status}</span>
                    </div>
                </div>
                
                <div class="job-info">
                    <div><strong>Frequency:</strong> Every ${job.frequency.value} ${job.frequency.unit}</div>
                    <div><strong>Last Run:</strong> ${lastRunText}</div>
                    <div><strong>Next Run:</strong> ${nextRunText}</div>
                    <div><strong>Logs:</strong> ${job.logs.length} executions</div>
                </div>
                
                <div class="job-actions">
                    <button class="btn btn-secondary btn-small" onclick="scheduler.viewLogs('${job.id}')">
                        📋 View Logs
                    </button>
                    ${job.status === 'pending' || job.status === 'running' ? 
                        `<button class="btn btn-danger btn-small" onclick="scheduler.cancelJob('${job.id}')">
                            ⏹️ Cancel
                        </button>` : ''
                    }
                    <button class="btn btn-danger btn-small" onclick="scheduler.deleteJob('${job.id}')">
                        🗑️ Delete
                    </button>
                </div>
            </div>
        `;
    }

    // Filtering and searching
    getFilteredJobs(jobs) {
        const searchTerm = document.getElementById('search-jobs').value.toLowerCase();
        const statusFilter = document.getElementById('filter-status').value;

        return jobs.filter(job => {
            const matchesSearch = job.name.toLowerCase().includes(searchTerm) ||
                                job.filename.toLowerCase().includes(searchTerm);
            const matchesStatus = !statusFilter || job.status === statusFilter;
            
            return matchesSearch && matchesStatus;
        }).sort((a, b) => new Date(b.createdAt) - new Date(a.createdAt));
    }

    filterJobs() {
        this.renderJobs();
    }

    // Utility methods
    generateId() {
        return Date.now().toString(36) + Math.random().toString(36).substr(2);
    }

    formatDateTime(dateString) {
        const date = new Date(dateString);
        return date.toLocaleString();
    }

    setDefaultDateTime() {
        const now = new Date();
        now.setMinutes(now.getMinutes() + 5); // Default to 5 minutes from now
        document.getElementById('start-time').value = now.toISOString().slice(0, 16);
    }

    updateJobsCount() {
        const count = this.jobs.size;
        document.getElementById('jobs-count').textContent = count;
    }

    resetForms() {
        document.getElementById('upload-form').reset();
        document.getElementById('scheduling-form').reset();
        document.getElementById('scheduling-section').classList.add('hidden');
        document.querySelector('.file-input-display').classList.remove('has-file');
        document.querySelector('.file-text').textContent = 'Choose .sh or .bat file (max 1MB)';
        document.getElementById('upload-btn').disabled = true;
        this.currentScript = null;
        this.setDefaultDateTime();
    }

    // UI helpers
    showError(element, message) {
        element.textContent = message;
        element.classList.remove('hidden');
    }

    showLoading(show) {
        const overlay = document.getElementById('loading-overlay');
        if (show) {
            overlay.classList.remove('hidden');
        } else {
            overlay.classList.add('hidden');
        }
    }

    showToast(message, type = 'info') {
        const container = document.getElementById('toast-container');
        const toast = document.createElement('div');
        
        const icons = {
            success: '✅',
            error: '❌',
            info: 'ℹ️'
        };

        toast.className = `toast ${type}`;
        toast.innerHTML = `
            <span class="toast-icon">${icons[type] || icons.info}</span>
            <span class="toast-message">${message}</span>
        `;

        container.appendChild(toast);

        // Auto remove after 5 seconds
        setTimeout(() => {
            if (toast.parentNode) {
                toast.parentNode.removeChild(toast);
            }
        }, 5000);
    }
}

const scheduler = new JobScheduler();

// Auto-refresh jobs view every 30 seconds
setInterval(() => {
    if (document.getElementById('jobs-tab').classList.contains('active')) {
        scheduler.renderJobs();
    }
}, 30000);
